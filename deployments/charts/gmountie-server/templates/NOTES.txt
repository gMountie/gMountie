gMountie server is running.

1. Get the gRPC endpoint:
{{- if .Values.gateway.enabled }}

   # Exposed via Gateway API (TLS on :443, gateway re-encrypts to the server):
   {{- range .Values.gateway.hostnames }}
   #   gmountie://{{ . }}:443
   {{- end }}
   {{- if not .Values.gateway.hostnames }}
   # (no gateway.hostnames set — clients connect on the Gateway's address)
   {{- end }}

{{- else if eq .Values.service.type "LoadBalancer" }}

   export GMOUNTIE_IP=$(kubectl -n {{ .Release.Namespace }} get svc {{ include "gmountie-server.fullname" . }} \
     -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
   echo "gmountie://${GMOUNTIE_IP}:{{ .Values.service.port }}"

{{- else if eq .Values.service.type "NodePort" }}

   export NODE_PORT=$(kubectl -n {{ .Release.Namespace }} get svc {{ include "gmountie-server.fullname" . }} \
     -o jsonpath='{.spec.ports[?(@.name=="grpc")].nodePort}')
   export NODE_IP=$(kubectl get nodes -o jsonpath='{.items[0].status.addresses[0].address}')
   echo "gmountie://${NODE_IP}:${NODE_PORT}"

{{- else }}

   kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "gmountie-server.fullname" . }} \
     {{ .Values.service.port }}:{{ .Values.service.port }}
   # then mount against 127.0.0.1:{{ .Values.service.port }}

{{- end }}

2. Mount from a FUSE-capable Linux client (endpoint from step 1):

   gmountie mount -s <host:port> -n {{ (index .Values.config.volumes 0).name }} \
{{- if .Values.auth.existingSecret }}
     -u <user from your auth Secret> \
     # (set -t <auth-type> if not basic; password via $GMOUNTIE_AUTH_PASSWORD)
{{- else if eq .Values.config.auth.type "basic" }}
     -u {{ (index .Values.config.auth.users 0).username }} \
     # password via $GMOUNTIE_AUTH_PASSWORD or the interactive prompt
{{- end }}
     /your/mountpoint

   # Or save the connection as a reusable profile
   # (~/.config/gmountie/profiles/<name>.yaml) and mount with:
   #   gmountie mount -P <name> -n {{ (index .Values.config.volumes 0).name }} /your/mountpoint

3. Health + metrics (ops HTTP endpoint):
{{- if include "gmountie-server.opsExposed" . }}

   # ops binds {{ include "gmountie-server.opsAddr" . }} (non-loopback) — reachable via the Service:
   kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "gmountie-server.fullname" . }} \
     {{ .Values.service.opsPort }}:{{ .Values.service.opsPort }}
   # auth is required on a non-loopback ops bind; with basic auth + TLS:
   curl -k -u <ops-user> https://127.0.0.1:{{ .Values.service.opsPort }}/healthz
   curl -k -u <ops-user> https://127.0.0.1:{{ .Values.service.opsPort }}/metrics

{{- else }}

   # ops binds {{ include "gmountie-server.opsAddr" . }} (loopback inside the pod). It is NOT
   # reachable through the Service; port-forward dials inside the pod netns:
   kubectl -n {{ .Release.Namespace }} port-forward deploy/{{ include "gmountie-server.fullname" . }} \
     {{ include "gmountie-server.opsPort" . }}:{{ include "gmountie-server.opsPort" . }}
   curl http://127.0.0.1:{{ include "gmountie-server.opsPort" . }}/healthz
   curl http://127.0.0.1:{{ include "gmountie-server.opsPort" . }}/readyz
   curl http://127.0.0.1:{{ include "gmountie-server.opsPort" . }}/metrics

{{- end }}
{{- if .Values.metrics.enabled }}

   # Prometheus scraping: the plain /metrics listener (no auth/TLS) is on
   # the `metrics` Service port ({{ .Values.metrics.port }}).
   curl http://<pod-or-svc>:{{ .Values.metrics.port }}/metrics
{{- if not .Values.metrics.serviceMonitor.enabled }}
   # Tip: metrics.serviceMonitor.enabled=true renders a ServiceMonitor for
   # Prometheus-Operator clusters.
{{- end }}
{{- else }}

   # For in-cluster Prometheus scraping without ops auth, enable the plain
   # /metrics listener: --set metrics.enabled=true (see values.yaml).
{{- end }}
